// The boot ELF (SLPM_662.80): Sony's libcdvd (version 3000, by its id string
// in .data at 0x195C00), text 0x1002A8-0x101F18; asm piece sdk/libcdvd.
// Sony's library code, kept as assembly: public API names only, where the
// binary shows them. Format: name = 0xADDRESS; // attributes

// The disc drive's start-up as Iop_Init calls it, named from the strings
// these calls use and the standard IOP reboot sequence.
sceCdInit = 0x001010D8; // type:func
sceCdMmode = 0x00101D60; // type:func

// Functions the game's C calls, named from what each does (its line's
// comment).
sceCdSync = 0x00100DC0; // type:func N command wait - mode 0 loops on sceSifCheckStatRpc with DelayThread (string N cmd wait)
sceCdDiskReady = 0x001015D8; // type:func string NEW DiskReady Call, falls back to the old DiskReady RPC
sceCdStandby = 0x00101AA8; // type:func N command RPC 6, no arguments, callback function code 5
sceCdPowerOff = 0x00101CA8; // type:func S command RPC 0x21, returns the status word through its pointer, called in the power-off loop after the devctl power-off calls
